  const issuedCount = totalBooks - availableBooks;
  const totalMembers = members.length;

  const bgColor = theme === 'light' ? 'bg-gray-50' : 'bg-gray-900';
  const cardBg = theme === 'light' ? 'bg-white' : 'bg-gray-800';
  const textColor = theme === 'light' ? 'text-gray-900' : 'text-gray-100';
  const textSecondary = theme === 'light' ? 'text-gray-600' : 'text-gray-400';
  const borderColor = theme === 'light' ? 'border-gray-200' : 'border-gray-700';
  const hoverBg = theme === 'light' ? 'hover:bg-gray-50' : 'hover:bg-gray-700';

  const openDialog = (type, item = null) => {
    setDialogType(type);
    setSelectedItem(item);
    setShowDialog(true);
  };

  const closeDialog = () => {
    setShowDialog(false);

    setSelectedItem(null);

  const StatCard = ({ icon: Icon, title, value, color }) => (

        <div className="flex gap-3">
          <button className="flex items-center gap-2 px-4 py-2 bg-gray-500 text-white rounded-lg hover:bg-gray-600">
            <Search className="w-4 h-4" />

          </button>

            onClick={() => openDialog('addBook')}
            className="flex items-center gap-2 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"
          >
            <Plus className="w-4 h-4" />
            Add Book
          </button>
        </div>
      </div>
      <div className={`${cardBg} rounded-lg shadow-md overflow-hidden border ${borderColor}`}>
        <table className="w-full">
          <thead className="bg-gray-100 dark:bg-gray-700">

              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Author</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Category</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Available</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Actions</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {books.map((book) => (
              <tr key={book.id} className={hoverBg}>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{book.id}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{book.title}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{book.author}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{book.category}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{book.available}/{book.totalCopies}</td>
                <td className="px-6 py-4 whitespace-nowrap text-sm">
                  <button 
                    onClick={() => openDialog('editBook', book)}
                    className="text-blue-500 hover:text-blue-700 mr-3"
                  >
                    <Edit2 className="w-4 h-4" />
                  </button>
                  <button

                </td>
              </tr>

          </tbody>
        </table>
      </div>
    </div>
  );

    <div className="p-6">
      <div className="flex justify-between items-center mb-6">
        <h1 className={`text-3xl font-bold ${textColor}`}>Members Management</h1>

            <Search className="w-4 h-4" />
            Search
          </button>
          <button 
            onClick={() => openDialog('addMember')}
            className="flex items-center gap-2 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"
          >
            <Plus className="w-4 h-4" />
            Add Member
          </button>
        </div>

      <div className={`${cardBg} rounded-lg shadow-md overflow-hidden border ${borderColor}`}>
        <table className="w-full">
          <thead className="bg-gray-100 dark:bg-gray-700">
            <tr>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Member ID</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Name</th>

              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Phone</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Books Issued</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Actions</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {members.map((member) => (
              <tr key={member.id} className={hoverBg}>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{member.id}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{member.name}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{member.contact}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{member.phone}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{member.booksIssued}</td>
                <td className="px-6 py-4 whitespace-nowrap text-sm">
                  <button 
                    onClick={() => openDialog('editMember', member)}
                    className="text-blue-500 hover:text-blue-700 mr-3"
                  >
                    <Edit2 className="w-4 h-4" />
                  </button>
                  <button 
                    onClick={() => openDialog('deleteMember', member)}
                    className="text-red-500 hover:text-red-700"
                  >
                    <Trash2 className="w-4 h-4" />
                  </button>
                </td>
              </tr>
            ))}
          </tbody>

    <div className="p-6">

      <div className={`${cardBg} rounded-lg shadow-md p-6 max-w-2xl border ${borderColor}`}>
        <div className="space-y-4">
          <div>
            <label className={`block text-sm font-medium ${textColor} mb-2`}>Select Book</label>
            <select className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}>

          </div>
          <div>
            <label className={`block text-sm font-medium ${textColor} mb-2`}>Select Member</label>
            <select className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}>

                <option key={member.id} value={member.id}>
                  {member.name} - {member.id} (Issued: {member.booksIssued})

              ))}
            </select>
          </div>
          <div>
            <label className={`block text-sm font-medium ${textColor} mb-2`}>Issue Date</label>
            <input 
              type="date" 
              className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}

            <label className={`block text-sm font-medium ${textColor} mb-2`}>Due Date</label>

            />
          </div>
          <button 
            onClick={() => openDialog('confirmIssue')}
            className="w-full mt-6 px-6 py-3 bg-blue-500 text-white rounded-lg hover:bg-blue-600 font-medium"
          >
            Issue Book
          </button>
        </div>
      </div>
    </div>
  );

  const ReturnBookScreen = () => (
    <div className="p-6">
      <h1 className={`text-3xl font-bold ${textColor} mb-6`}>Return Book</h1>
      <div className={`${cardBg} rounded-lg shadow-md overflow-hidden border ${borderColor}`}>
        <table className="w-full">
          <thead className="bg-gray-100 dark:bg-gray-700">
            <tr>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Issue ID</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Book Title</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Member Name</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Issue Date</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Due Date</th>
              <th className={`px-6 py-3 text-left text-xs font-medium ${textSecondary} uppercase tracking-wider`}>Action</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {issuedBooks.map((issue) => (
              <tr key={issue.issueId} className={hoverBg}>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{issue.issueId}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{issue.bookTitle}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{issue.memberName}</td>
                <td className={`px-6 py-4 whitespace-nowrap text-sm ${textColor}`}>{issue.issueDate}</td>

                <td className="px-6 py-4 whitespace-nowrap text-sm">
                  <button 
                    onClick={() => openDialog('confirmReturn', issue)}
                    className="px-4 py-2 bg-green-500 text-white rounded-lg hover:bg-green-600"
                  >
                    Return
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>

    <div className="p-6">
      <h1 className={`text-3xl font-bold ${textColor} mb-6`}>Reports</h1>
      <div className="space-y-6">
        <div className={`${cardBg} rounded-lg shadow-md p-6 border ${borderColor}`}>
          <h2 className={`text-xl font-semibold ${textColor} mb-4`}>Issued Books Report</h2>
          <p className={`${textSecondary} mb-4`}>Total Issued Books: {issuedCount}</p>

            <table className="w-full">

                  <th className={`px-4 py-2 text-left text-xs font-medium ${textSecondary} uppercase`}>Issue Date</th>
                </tr>
              </thead>
              <tbody>
                {issuedBooks.map((issue, idx) => (

                    <td className={`px-4 py-2 text-sm ${textColor}`}>{issue.bookTitle}</td>
                    <td className={`px-4 py-2 text-sm ${textColor}`}>{issue.memberName}</td>
                    <td className={`px-4 py-2 text-sm ${textColor}`}>{issue.issueDate}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

        <div className={`${cardBg} rounded-lg shadow-md p-6 border ${borderColor}`}>
          <h2 className={`text-xl font-semibold ${textColor} mb-4`}>Available Books Report</h2>
          <p className={`${textSecondary} mb-4`}>Total Available: {availableBooks}</p>
          <div className="overflow-auto">
            <table className="w-full">
              <thead className="bg-gray-100 dark:bg-gray-700">
                <tr>
                  <th className={`px-4 py-2 text-left text-xs font-medium ${textSecondary} uppercase`}>Title</th>
                  <th className={`px-4 py-2 text-left text-xs font-medium ${textSecondary} uppercase`}>Author</th>
                  <th className={`px-4 py-2 text-left text-xs font-medium ${textSecondary} uppercase`}>Available</th>
                </tr>
              </thead>
              <tbody>
                {books.filter(b => b.available > 0).map((book, idx) => (
                  <tr key={idx} className={`border-b ${borderColor}`}>
                    <td className={`px-4 py-2 text-sm ${textColor}`}>{book.title}</td>
                    <td className={`px-4 py-2 text-sm ${textColor}`}>{book.author}</td>
                    <td className={`px-4 py-2 text-sm ${textColor}`}>{book.available}</td>
                  </tr>
                ))}
              </tbody>

          </div>

    const getDialogContent = () => {
      switch (dialogType) {
        case 'addBook':
        case 'editBook':
          return (
            <div className="space-y-4">
              <h2 className={`text-2xl font-bold ${textColor} mb-4`}>
                {dialogType === 'addBook' ? 'Add New Book' : 'Edit Book'}
              </h2>
              <input 
                type="text" 
                placeholder="Book ID" 
                defaultValue={selectedItem?.id || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <input 
                type="text" 
                placeholder="Title" 
                defaultValue={selectedItem?.title || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <input

                placeholder="Author" 
                defaultValue={selectedItem?.author || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}

                type="text"

                type="number"

                <button

                  className="flex-1 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"
                >
                  {dialogType === 'addBook' ? 'Add' : 'Update'}
                </button>
              </div>
            </div>
          );

        case 'addMember':
        case 'editMember':
          return (
            <div className="space-y-4">
              <h2 className={`text-2xl font-bold ${textColor} mb-4`}>
                {dialogType === 'addMember' ? 'Add New Member' : 'Edit Member'}

              <input 
                type="text" 
                placeholder="Member ID" 
                defaultValue={selectedItem?.id || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <input 
                type="text" 
                placeholder="Name" 
                defaultValue={selectedItem?.name || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <input 
                type="email" 
                placeholder="Email" 
                defaultValue={selectedItem?.contact || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <input 
                type="tel" 
                placeholder="Phone" 
                defaultValue={selectedItem?.phone || ''}
                className={`w-full px-4 py-2 border ${borderColor} rounded-lg ${cardBg} ${textColor}`}
              />
              <div className="flex gap-3 mt-6">
                <button 
                  onClick={closeDialog}
                  className="flex-1 px-4 py-2 bg-gray-500 text-white rounded-lg hover:bg-gray-600"

                  Cancel
                </button>
                <button 
                  onClick={closeDialog}
                  className="flex-1 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"

                  {dialogType === 'addMember' ? 'Add' : 'Update'}
                </button>
              </div>
            </div>

        case 'deleteBook':
        case 'deleteMember':

            <div className="space-y-4">
              <h2 className={`text-2xl font-bold ${textColor} mb-4`}>Confirm Deletion</h2>

              <h2 className={`text-2xl font-bold ${textColor} mb-4`}>Confirm Return</h2>

                </button>
                <button 
                  onClick={closeDialog}
                  className="flex-1 px-4 py-2 bg-green-500 text-white rounded-lg hover:bg-green-600"

              </div>

          <NavButton icon={Home} label="Dashboard" screen="dashboard" />
          <NavButton icon={Book} label="Books" screen="books" />
          <NavButton icon={Users} label="Members" screen="members" />
          <NavButton icon={BookOpen} label="Issue Book" screen="issue" />
          <NavButton icon={BookCheck} label="Return Book" screen="return" />
          <NavButton icon={FileText} label="Reports" screen="reports" />
        </nav>

        <div className="p-4 border-t border-gray-200 dark:border-gray-700">
          <button
            onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}
            className={`w-full flex items-center justify-center gap-2 px-4 py-3 rounded-lg ${hoverBg} ${textColor}`}
          >
            {theme === 'light' ? '🌙' : '☀️'}
            <span>Toggle Theme</span>
          </button>
          <button
            className={`w-full flex items-center justify-center gap-2 px-4 py-3 rounded-lg ${hoverBg} ${textColor} mt-2`}
          >
            <LogOut className="w-5 h-5" />
            <span>Exit</span>
          </button>
        </div>
      </div>

      <div className="flex-1 overflow-y-auto">
        {activeScreen === 'dashboard' && <DashboardScreen />}
        {activeScreen === 'books' && <BooksScreen />}
        {activeScreen === 'members' && <MembersScreen />}
        {activeScreen === 'issue' && <IssueBookScreen />}
        {activeScreen === 'return' && <ReturnBookScreen />}
        {activeScreen === 'reports' && <ReportsScreen />}
      </div>

      <Dialog />
    </div>
  );
};

export default LibraryManagementSystem;
